#  COMBINED DUELS + PER-GAMEMODE MMR SYSTEM
#  8 ranked gamemodes: sword, axe, dpot, nethpot, smp, crystal, mace, uhc
#  Each mode uses MMR (Elo-style) with tier boundaries:
#  LT5 0-499 | HT5 500-899 | LT4 900-1299 | HT4 1300-1699
#  LT3 1700-2099 | HT3 2100-2499 | LT2 2500-2899 | HT2 2900-3299
#  LT1 3300-3699 | HT1 — held by defeating the current HT1
#  /rank shows MMR + tier in every gamemode
#
#  PLACEMENT: 10 matches per gamemode required before rank is shown
#  DETERIORATION: LT2+ players lose MMR if inactive for 3+ days
#
#  ROUNDS: FT1–FT15 selectable per duel. MMR only applied when the
#  series winner reaches the FT target. Between rounds players are
#  rekitted and teleported back to arena spawn points automatically.
# ============================================================

function getModeMMR(p: player, mode: text) :: number:
    set {_uuid} to "%uuid of {_p}%"
    if {mmr.%{_mode}%::%{_uuid}%} is set:
        return {mmr.%{_mode}%::%{_uuid}%}
    return 1000

function getModeFights(p: player, mode: text) :: number:
    set {_uuid} to "%uuid of {_p}%"
    if {fights.%{_mode}%::%{_uuid}%} is set:
        return {fights.%{_mode}%::%{_uuid}%}
    return 0

function hasCompletedPlacements(uuid: text, mode: text) :: boolean:
    if {fights.%{_mode}%::%{_uuid}%} is set:
        if {fights.%{_mode}%::%{_uuid}%} >= 10:
            return true
    return false

function tierFromMMR(mmr: number) :: text:
    if {_mmr} < 500:
        return "LT5"
    else if {_mmr} < 900:
        return "HT5"
    else if {_mmr} < 1300:
        return "LT4"
    else if {_mmr} < 1700:
        return "HT4"
    else if {_mmr} < 2100:
        return "LT3"
    else if {_mmr} < 2500:
        return "HT3"
    else if {_mmr} < 2900:
        return "LT2"
    else if {_mmr} < 3300:
        return "HT2"
    else:
        return "LT1"

function getModeRank(uuid: text, mode: text) :: text:
    if {ht1holder.%{_mode}%} is {_uuid}:
        return "HT1"
    if hasCompletedPlacements({_uuid}, {_mode}) is false:
        return "Unranked"
    if {mmr.%{_mode}%::%{_uuid}%} is set:
        return tierFromMMR({mmr.%{_mode}%::%{_uuid}%})
    return tierFromMMR(1000)

function tierProgress(mmr: number, uuid: text, mode: text) :: number:
    if {ht1holder.%{_mode}%} is {_uuid}:
        return 100
    if {_mmr} < 500:
        return round(({_mmr} / 500) * 100)
    else if {_mmr} < 900:
        return round((({_mmr} - 500) / 400) * 100)
    else if {_mmr} < 1300:
        return round((({_mmr} - 900) / 400) * 100)
    else if {_mmr} < 1700:
        return round((({_mmr} - 1300) / 400) * 100)
    else if {_mmr} < 2100:
        return round((({_mmr} - 1700) / 400) * 100)
    else if {_mmr} < 2500:
        return round((({_mmr} - 2100) / 400) * 100)
    else if {_mmr} < 2900:
        return round((({_mmr} - 2500) / 400) * 100)
    else if {_mmr} < 3300:
        return round((({_mmr} - 2900) / 400) * 100)
    else:
        return round((({_mmr} - 3300) / 400) * 100)

function expectedWin(mmrA: number, mmrB: number) :: number:
    set {_diff} to ({_mmrB} - {_mmrA}) / 400
    set {_pow} to 10 ^ {_diff}
    return 1.0 / (1.0 + {_pow})

function calcGain(winnerMMR: number, loserMMR: number, winnerFights: number) :: number:
    set {_k} to 120
    if {_winnerFights} < 10:
        set {_k} to 300
    set {_expected} to expectedWin({_winnerMMR}, {_loserMMR})
    return round({_k} * (1.0 - {_expected}))

function calcLoss(loserMMR: number, winnerMMR: number, loserFights: number) :: number:
    set {_k} to 120
    if {_loserFights} < 10:
        set {_k} to 300
    set {_expected} to expectedWin({_loserMMR}, {_winnerMMR})
    return round({_k} * {_expected})

function applyDuelMMR(winner: player, loser: player, mode: text):
    set {_wUUID} to "%uuid of {_winner}%"
    set {_lUUID} to "%uuid of {_loser}%"

    set {_wMMR} to getModeMMR({_winner}, {_mode})
    set {_lMMR} to getModeMMR({_loser}, {_mode})

    set {_lFights} to getModeFights({_loser}, {_mode})

    set {_loss} to calcLoss({_lMMR}, {_wMMR}, {_lFights})

    set {_oldLRank} to getModeRank({_lUUID}, {_mode})

    set {_newLMMR} to min(9999, max(0, {_lMMR} - {_loss}))

    set {mmr.%{_mode}%::%{_lUUID}%} to {_newLMMR}

    if {ht1holder.%{_mode}%} is not set:
        if {_newWMMR} >= 3700:
            if hasCompletedPlacements({_wUUID}, {_mode}) is true:
                set {ht1holder.%{_mode}%} to {_wUUID}

function syncPlayerToWebsite(p: player, mode: text):
    if {duels.db} is not set:
        stop
    set {_uuid} to "%uuid of {_p}%"
    set {_name} to name of {_p}
    set {_ip} to ip of {_p}
    set {_mmr} to getModeMMR({_p}, {_mode})
    set {_fights} to getModeFights({_p}, {_mode})
    set {_wins} to {wins.%{_mode}%::%{_uuid}%}
    if {_wins} is not set:
        set {_wins} to 0
    set {_losses} to {losses.%{_mode}%::%{_uuid}%}
    if {_losses} is not set:
        set {_losses} to 0
    execute unsafe "INSERT INTO players (uuid, username, ip, region_checked) VALUES ('%{_uuid}%', '%{_name}%', '%{_ip}%', 0) ON DUPLICATE KEY UPDATE username='%{_name}%', ip='%{_ip}%', region_checked=0" in {duels.db}
    execute unsafe "INSERT INTO player_stats (uuid, mode, mmr, wins, losses, fights, last_fight) VALUES ('%{_uuid}%', '%{_mode}%', %{_mmr}%, %{_wins}%, %{_losses}%, %{_fights}%, NOW()) ON DUPLICATE KEY UPDATE mmr=%{_mmr}%, wins=%{_wins}%, losses=%{_losses}%, fights=%{_fights}%, last_fight=NOW()" in {duels.db}
    if {ht1holder.%{_mode}%} is {_uuid}:
        execute unsafe "INSERT INTO ht1_holders (mode, uuid) VALUES ('%{_mode}%', '%{_uuid}%') ON DUPLICATE KEY UPDATE uuid='%{_uuid}%'" in {duels.db}
    else:
        execute unsafe "DELETE FROM ht1_holders WHERE mode='%{_mode}%' AND uuid='%{_uuid}%'" in {duels.db}

function getTotalMMR(uuid: text) :: number:
    set {_total} to 0
    set {_rankedModes} to 0
    loop "sword", "axe", "dpot", "nethpot", "smp", "crystal", "mace", "uhc":
        if hasCompletedPlacements({_uuid}, loop-value) is true:
            if {mmr.%loop-value%::%{_uuid}%} is set:
                add {mmr.%loop-value%::%{_uuid}%} to {_total}
            else:
                add 1000 to {_total}
            add 1 to {_rankedModes}
    if {_rankedModes} is 0:
        return 0
    return {_total}

function getOverallRank(uuid: text) :: number:
    set {_myTotal} to getTotalMMR({_uuid})
    set {_pos} to 1
    loop "sword", "axe", "dpot", "nethpot", "smp", "crystal", "mace", "uhc":
        loop {mmr.%loop-value%::*}:
            set {_otherUUID} to loop-index-1
            if {_otherUUID} is not {_uuid}:
                if {_seen::%{_otherUUID}%} is not true:
                    set {_seen::%{_otherUUID}%} to true
                    if getTotalMMR({_otherUUID}) > {_myTotal}:
                        add 1 to {_pos}
    return {_pos}

    set {_mode}      to {duel.type::%{_p1}%}

    set {_p2W}       to {duel.roundwins::%{_p2}%}

function decayMode(mode: text, decayAmount: number, decayFloor: number):
    loop {mmr.%{_mode}%::*}:
        set {_uuid} to loop-index-1
        set {_mmr} to loop-value-1
        if {_mmr} >= {_decayFloor}:
            if {mmr.lastfight.%{_mode}%::%{_uuid}%} is set:
                set {_lastFight} to {mmr.lastfight.%{_mode}%::%{_uuid}%}
                if time since {_lastFight} > 3 days:
                    set {_newMMR} to max({_decayFloor}, {_mmr} - {_decayAmount})
                    set {mmr.%{_mode}%::%{_uuid}%} to {_newMMR}
                    if {ht1holder.%{_mode}%} is {_uuid}:
                        if {_newMMR} < 3300:
                            delete {ht1holder.%{_mode}%}
                            loop all players:
                                if "%uuid of loop-player%" is {_uuid}:
                                    send "{@prefix} &cYour HT1 throne in &f%{_mode}%&c has been vacated due to inactivity." to loop-player
            else:
                set {mmr.lastfight.%{_mode}%::%{_uuid}%} to now

function getFreeSmallArena() :: number:
    loop integers from 1 to 7:
        if {arena.small.%loop-number%.inuse} is not true:
            return loop-number
    return 0

function getFreeBigArena() :: number:
    loop integers from 1 to 3:
        if {arena.big.%loop-number%.inuse} is not true:
            return loop-number
    return 0

function saveDuelState(p1: player, p2: player):
    loop 36 times:
        set {saved.inv.%{_p1}%.%loop-number - 1%} to slot (loop-number - 1) of {_p1}'s inventory
    set {saved.armor.helmet::%{_p1}%}     to {_p1}'s helmet
    set {saved.armor.chestplate::%{_p1}%} to {_p1}'s chestplate
    set {saved.armor.leggings::%{_p1}%}   to {_p1}'s leggings
    set {saved.armor.boots::%{_p1}%}      to {_p1}'s boots
    set {saved.armor.offhand::%{_p1}%}    to {_p1}'s offhand tool
    set {duel.pos::%{_p1}%}               to location of {_p1}
    loop 36 times:
        set {saved.inv.%{_p2}%.%loop-number - 1%} to slot (loop-number - 1) of {_p2}'s inventory
    set {saved.armor.helmet::%{_p2}%}     to {_p2}'s helmet
    set {saved.armor.chestplate::%{_p2}%} to {_p2}'s chestplate
    set {saved.armor.leggings::%{_p2}%}   to {_p2}'s leggings
    set {saved.armor.boots::%{_p2}%}      to {_p2}'s boots
    set {saved.armor.offhand::%{_p2}%}    to {_p2}'s offhand tool
    set {duel.pos::%{_p2}%}               to location of {_p2}

function clearForDuel(p: player):
    clear {_p}'s inventory
    set {_p}'s helmet     to air
    set {_p}'s chestplate to air
    set {_p}'s leggings   to air
    set {_p}'s boots      to air
    set {_p}'s offhand tool to air

function restorePlayer(p: player):
    clear {_p}'s inventory
    set {_p}'s helmet     to air
    set {_p}'s chestplate to air
    set {_p}'s leggings   to air
    set {_p}'s boots      to air
    set {_p}'s offhand tool to air
    loop 36 times:
        set slot (loop-number - 1) of {_p}'s inventory to {saved.inv.%{_p}%.%loop-number - 1%}
    delete {saved.inv.%{_p}%::*}
    set {_p}'s helmet     to {saved.armor.helmet::%{_p}%}
    set {_p}'s chestplate to {saved.armor.chestplate::%{_p}%}
    set {_p}'s leggings   to {saved.armor.leggings::%{_p}%}
    set {_p}'s boots      to {saved.armor.boots::%{_p}%}
    set {_p}'s offhand tool to {saved.armor.offhand::%{_p}%}
    delete {saved.armor.helmet::%{_p}%}
    delete {saved.armor.chestplate::%{_p}%}
    delete {saved.armor.leggings::%{_p}%}
    delete {saved.armor.boots::%{_p}%}
    delete {saved.armor.offhand::%{_p}%}
    teleport {_p} to {duel.pos::%{_p}%}
    delete {duel.pos::%{_p}%}

function discardSavedState(p: player):
    delete {saved.inv.%{_p}%::*}
    delete {saved.armor.helmet::%{_p}%}
    delete {saved.armor.chestplate::%{_p}%}
    delete {saved.armor.leggings::%{_p}%}
    delete {saved.armor.boots::%{_p}%}
    delete {saved.armor.offhand::%{_p}%}
    delete {duel.pos::%{_p}%}

function cleanDuelVars(p: player):
    delete {duel.active::%{_p}%}
    delete {duel.opponent::%{_p}%}
    delete {duel.type::%{_p}%}
    delete {duel.arena::%{_p}%}
    delete {duel.arenatype::%{_p}%}
    delete {duel.shulker.speed::%{_p}%}
    delete {duel.shulker.strength::%{_p}%}

command /duel <player>:
    trigger:
        if arg-1 is player:
            send "{@prefix} &cYou cannot duel yourself." to player
            stop

        if arg-1 is not online:
            send "{@prefix} &cPlayer not online." to player
            stop

        if {duel.active::%arg-1%} is true:
            send "{@prefix} &cThat player is already in a duel." to player
            stop

        if {duel.pending::%arg-1%} is set:
            send "{@prefix} &cThat player already has a pending duel request." to player
            stop
        if {duel.request::%player%} is set:
            send "{@prefix} &cYou already have a pending duel request." to player
            stop
        if {duel.active::%player%} is true:
            send "{@prefix} &cYou are already in a duel." to player
            stop

        set {duel.request::%player%} to arg-1
        set {duel.pending::%arg-1%} to player

        send "{@prefix} &e%arg-1% &ahas been sent a duel request by &e%player%&a." to player
        send "{@prefix} &e%player% &awants to duel you! Type &e/acceptduel %player% &ato accept or &c/denyduel &cto deny." to arg-1

        wait 30 seconds

        if {duel.pending::%arg-1%} is player:
            delete {duel.pending::%arg-1%}
            delete {duel.request::%player%}
            send "{@prefix} &cYour duel request to &e%arg-1% &chas expired." to player
command /resetrequest <player>:
    permission: op
    trigger:
        delete {duel.request::%arg-1%}
        delete {duel.request::%player%}
# ============================================================
#  COMMAND: /acceptduel <player>
# ============================================================
command /acceptduel <player>:
    trigger:
        if {duel.pending::%player%} is not set:
            send "{@prefix} &cYou have no pending duel request." to player
            stop

        if {duel.pending::%player%} is not arg-1:
            send "{@prefix} &cYou don't have a pending duel request from &e%arg-1%&c." to player
            stop

        set {_requester} to {duel.pending::%player%}

        if {_requester} is not online:
            send "{@prefix} &cThe player who challenged you is no longer online." to player
            delete {duel.pending::%player%}
            stop

        if {duel.active::%player%} is true:
            send "{@prefix} &cYou are already in a duel." to player
            delete {duel.pending::%player%}
            stop

        delete {duel.pending::%player%}

        set {_gui} to chest inventory with 4 rows named "&bSelect Kit"

        set slot 11 of {_gui} to diamond sword named "&9Sword"
        set slot 12 of {_gui} to diamond axe named "&3Axe"
        set slot 13 of {_gui} to potion of instant health named "&bDiamond Pot"
        set slot 14 of {_gui} to netherite helmet of protection 4 named "&8Netherite Pot"
        set slot 15 of {_gui} to shield named "&1SMP"
        set slot 20 of {_gui} to chest named "&eYour Inventory"
        set slot 21 of {_gui} to end crystal named "&dCrystalPVP"
        set slot 22 of {_gui} to mace named "&5Mace"
        set slot 23 of {_gui} to red dye named "&6UHC"
        set slot 8 of {_gui} to paper named "&eSet Round Count &7(click)"

        send "{@prefix} &e%player% &aaccepted the duel! Select a kit, then set a round count." to {_requester}
        send "{@prefix} &aSelect a kit, then set a round count for the duel!" to player

        open {_gui} to {_requester}
        set {duel.setup::%{_requester}%} to true
        
# ============================================================
#  COMMAND: /denyduel
# ============================================================
command /denyduel:
    trigger:
        if {duel.pending::%player%} is not set:
            send "{@prefix} &cYou have no pending duel request." to player
            stop

        set {_requester} to {duel.pending::%player%}
        delete {duel.pending::%player%}
        delete {duel.request::%{_requester}%}

        send "{@prefix} &e%player% &cdenied your duel request." to {_requester}
        send "{@prefix} &cYou denied the duel request." to player

# ============================================================
#  COMMAND: /endduel  — forfeiter is treated as the loser
# ============================================================
command /endduel:
    trigger:
        if {duel.active::%player%} is not true:
            send "{@prefix} &cYou are not in a duel." to player
            stop

        set {_loser}     to player
        set {_winner}    to {duel.opponent::%player%}
        set {_type}      to {duel.type::%player%}
        set {_arena}     to {duel.arena::%player%}
        set {_arenaType} to {duel.arenatype::%player%}

        send "{@prefix} &e%{_loser}% &chas forfeited the duel." to {_loser}

        if {_type} is not "custom":
            if {_winner} is a player:
                applyDuelMMR({_winner}, {_loser}, {_type})

        if {_winner} is set:
            send "{@prefix} &e%{_loser}% &cforfeited — &e%{_winner}% &awins the series!" to {_winner}

        restorePlayer({_loser})
        cleanDuelVars({_loser})
        cleanRoundVars({_loser})

        if {_winner} is set:
            restorePlayer({_winner})
            cleanDuelVars({_winner})
            cleanRoundVars({_winner})

        if {_arena} is set:
            if {_arenaType} is "big":
                delete {arena.big.%{_arena}%.inuse}
            else:
                delete {arena.small.%{_arena}%.inuse}

on chat:

        if message is "cancel":

            delete {duel.awaitingft::%{_p1}%}

            send "{@prefix} &cInvalid number — type a number between 1 and 15." to {_p1}
            stop
        if {_ft} < 1:
            send "{@prefix} &cMinimum round count is 1." to {_p1}
            stop
        if {_ft} > 15:
            send "{@prefix} &cMaximum round count is 15." to {_p1}
            stop

#  DEATH (DUEL) — ROUND LOGIC HERE, MMR ONLY ON SERIES WIN
# ============================================================
on death:
    if {duel.active::%victim%} is true:
        clear drops
        set {_loser}     to victim
        set {_winner}    to {duel.opponent::%{_loser}%}
        set {_type}      to {duel.type::%{_loser}%}
        set {_arena}     to {duel.arena::%{_loser}%}
        set {_arenaType} to {duel.arenatype::%{_loser}%}
        set {_ft}        to {duel.ft::%{_loser}%}

        delete {duel.active::%{_loser}%}

        # ── TIE CHECK ────────────────────────────────────────────
        if {duel.active::%{_winner}%} is not true:
            send "{@prefix} &eIt's a tie! Both players died at the same time." to {_loser}
            if {_winner} is online:
                send "{@prefix} &eIt's a tie! Both players died at the same time." to {_winner}

            if {_arena} is set:
                if {_arenaType} is "big":
                    delete {arena.big.%{_arena}%.inuse}
                else:
                    delete {arena.small.%{_arena}%.inuse}

            cleanDuelVars({_loser})
            cleanRoundVars({_loser})
            set {duel.loser::%{_loser}%} to true

            if {_winner} is online:
                set {duel.pendingwin::%{_winner}%} to true
            stop

        # ── ROUND WIN ─────────────────────────────────────────────
        delete {duel.active::%{_winner}%}
        add 1 to {duel.roundwins::%{_winner}%}

        set {_wWins} to {duel.roundwins::%{_winner}%}
        set {_lWins} to {duel.roundwins::%{_loser}%}

        # ── SERIES OVER? ─────────────────────────────────────────
        if {_wWins} >= {_ft}:
            send "{@prefix} &e%{_winner}% &awon the series &a(%{_wWins}%-%{_lWins}%) against &e%{_loser}%&a!" to {_winner}
            send "{@prefix} &cYou lost the series &7(%{_lWins}%-%{_wWins}%) against &e%{_winner}%&c!" to {_loser}

            if {_type} is not "custom":
                if {_winner} is a player:
                    if {_loser} is a player:
                        applyDuelMMR({_winner}, {_loser}, {_type})

            if {_arena} is set:
                if {_arenaType} is "big":
                    delete {arena.big.%{_arena}%.inuse}
                else:
                    delete {arena.small.%{_arena}%.inuse}

            cleanRoundVars({_winner})
            cleanRoundVars({_loser})
            set {duel.loser::%{_loser}%} to true
            set {duel.pendingwin::%{_winner}%} to true

        else:
            # ── NEXT ROUND ────────────────────────────────────────
            send "" to {_winner}
            send "&6&lRound over! &e%{_winner}% &awon that round. &7Score: &a%{_wWins}%&7-&c%{_lWins}% &7(FT%{_ft}%)" to {_winner}
            send "" to {_winner}
            send "" to {_loser}
            send "&6&lRound over! &e%{_winner}% &awon that round. &7Score: &a%{_wWins}%&7-&c%{_lWins}% &7(FT%{_ft}%)" to {_loser}
            send "" to {_loser}
            heal {_winner}
            execute console command "effect clear %{_winner}%"
            set {duel.inroundbreak::%{_winner}%} to true
            set {duel.inroundbreak::%{_loser}%}  to true

            set {duel.loser::%{_loser}%} to true
            set {duel.pendingwin::%{_winner}%} to true

            loop 1 times:
                set {_countdown} to 6 - loop-number
                wait 20 ticks
                if {_winner} is online:
                    send action bar "&eNext round in &c%{_countdown}%&e..." to {_winner}
                if {_loser} is online:
                    send action bar "&eNext round in &c%{_countdown}%&e..." to {_loser}

            if {_winner} is online:
                if {_loser} is online:
                    delete {duel.inroundbreak::%{_winner}%}
                    delete {duel.inroundbreak::%{_loser}%}
                    delete {duel.loser::%{_loser}%}
                    delete {duel.pendingwin::%{_winner}%}
                    startNextRound({_winner}, {_loser})
                else:
                    send "{@prefix} &cYour opponent disconnected — you win the series!" to {_winner}
                    if {_type} is not "custom":
                        applyDuelMMR({_winner}, {_loser}, {_type})
                    if {_arena} is set:
                        if {_arenaType} is "big":
                            delete {arena.big.%{_arena}%.inuse}
                        else:
                            delete {arena.small.%{_arena}%.inuse}
                    restorePlayer({_winner})
                    cleanDuelVars({_winner})
                    cleanRoundVars({_winner})
                    delete {duel.inroundbreak::%{_winner}%}
                    delete {duel.pendingwin::%{_winner}%}

every 3 ticks:

function getPointsFromRank(rank: text) :: number:
    if {_rank} is "HT1":
        return 60
    else if {_rank} is "LT1":
        return 45
    else if {_rank} is "HT2":
        return 30
    else if {_rank} is "LT2":
        return 20
    else if {_rank} is "HT3":
        return 10
    else if {_rank} is "LT3":
        return 6
    else if {_rank} is "HT4":
        return 4
    else if {_rank} is "LT4":
        return 3
    else if {_rank} is "HT5":
        return 2
    else if {_rank} is "LT5":
        return 1
    return 0

function getTotalPoints(uuid: text) :: number:
    set {_total} to 0
    loop "sword", "axe", "dpot", "nethpot", "smp", "crystal", "mace", "uhc":
        if hasCompletedPlacements({_uuid}, loop-value) is true:
            set {_rank} to getModeRank({_uuid}, loop-value)
            add getPointsFromRank({_rank}) to {_total}
    return {_total}

                set {mmr.%loop-value%::%{_uuid}%} to 1000
                set {fights.%loop-value%::%{_uuid}%} to 0

                if {ht1holder.%loop-value%} is {_uuid}:
                    delete {ht1holder.%loop-value%}
