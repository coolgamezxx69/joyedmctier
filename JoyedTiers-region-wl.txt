# JoyedTiers — Wins, Losses & Region Tracking
# ------------------------------------------------------------
# SETUP INSTRUCTIONS:
#
# 1. Run these SQL commands ONCE in your PebbleHost MySQL panel:
#
#    ALTER TABLE player_stats ADD COLUMN wins INT NOT NULL DEFAULT 0;
#    ALTER TABLE player_stats ADD COLUMN losses INT NOT NULL DEFAULT 0;
#    ALTER TABLE players ADD COLUMN region VARCHAR(2) NOT NULL DEFAULT 'US';
#
# 2. Set the region option below to US or EU for this server.
#
# 3. If you have separate US and EU servers, use this file on both
#    but change the region option on each one.
#
# 4. Replace {joyedtiers.db} with your actual sk-db variable name
#    (the same one used in your main JoyedTiers Skript).
#
# 5. Drop this file into your Skript plugins/Skript/scripts/ folder
#    and run /skript reload JoyedTiers-region-wl
# ============================================================

options:
    # Change to EU if this is your European server
    region: US
    db: {joyedtiers.db}

# ------------------------------------------------------------
# On load: stamp all existing players with this server's region
# ------------------------------------------------------------
on skript load:
    execute "UPDATE players SET region = '{@region}'" in {@db}
    broadcast "&8[&6JoyedTiers&8] &7Region set to &b{@region} &7for all players."

# ------------------------------------------------------------
# On join: make sure new players also get the region
# ------------------------------------------------------------
on join:
    wait 2 seconds
    set {_uuid} to uuid of player
    execute "UPDATE players SET region = '{@region}' WHERE uuid = '%{_uuid}%'" in {@db}

# ------------------------------------------------------------
# recordDuelResult(winner, loser, mode)
# Call this function from your existing arena end event,
# replacing wherever you currently call your fight/mmr update.
#
# Example — in your arena end code replace:
#   [your old fight recording code here]
# with:
#   recordDuelResult(winner, loser, "sword")
# ------------------------------------------------------------
function recordDuelResult(winner: player, loser: player, mode: text):
    set {_wu} to uuid of {_winner}
    set {_lu} to uuid of {_loser}

    # ---- Record WIN for winner ----
    execute "SELECT COUNT(*) as cnt FROM player_stats WHERE uuid = '%{_wu}%' AND mode = '%{_mode}%'" in {@db}
    set {_rows::*} to last database result
    set {_cnt} to {_rows::1::cnt} parsed as integer
    if {_cnt} > 0:
        execute "UPDATE player_stats SET wins = wins + 1, fights = fights + 1 WHERE uuid = '%{_wu}%' AND mode = '%{_mode}%'" in {@db}
    else:
        execute "INSERT INTO player_stats (uuid, mode, mmr, wins, losses, fights) VALUES ('%{_wu}%', '%{_mode}%', 1000, 1, 0, 1)" in {@db}

    # ---- Record LOSS for loser ----
    execute "SELECT COUNT(*) as cnt FROM player_stats WHERE uuid = '%{_lu}%' AND mode = '%{_mode}%'" in {@db}
    set {_rows2::*} to last database result
    set {_cnt2} to {_rows2::1::cnt} parsed as integer
    if {_cnt2} > 0:
        execute "UPDATE player_stats SET losses = losses + 1, fights = fights + 1 WHERE uuid = '%{_lu}%' AND mode = '%{_mode}%'" in {@db}
    else:
        execute "INSERT INTO player_stats (uuid, mode, mmr, wins, losses, fights) VALUES ('%{_lu}%', '%{_mode}%', 1000, 0, 1, 1)" in {@db}

    # ---- Ensure both players have region ----
    execute "UPDATE players SET region = '{@region}' WHERE uuid IN ('%{_wu}%', '%{_lu}%')" in {@db}

# ------------------------------------------------------------
# setPlayerMMR(player, mode, newMMR)
# Use this when you update a player's MMR (after ELO calc).
# This does NOT change wins/losses — just the MMR value.
# ------------------------------------------------------------
function setPlayerMMR(p: player, mode: text, newMMR: integer):
    set {_uuid} to uuid of {_p}
    execute "UPDATE player_stats SET mmr = %{_newMMR}% WHERE uuid = '%{_uuid}%' AND mode = '%{_mode}%'" in {@db}
